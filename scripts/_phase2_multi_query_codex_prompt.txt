2. v1/retrieval/multi_query.py — NEW
   Self-contained module. No dependency on v1/main.py.

       """Multi-query expansion (T02) for Phase 2 vocabulary bridging.

       Calls JUDGE_MODEL to generate N alternative phrasings of the user
       query, runs section-level retrieval for each, unions sections by
       (page_rel, section_title), keeps the max score per section, sorts,
       and formats with the existing token-budget assembler.
       """

       from __future__ import annotations
       from dataclasses import dataclass
       from v1.retrieval.wiki_retriever import retrieve_sections, format_sections
       from v1.retrieval.wiki_chunker import WikiSection
       from shared.llm_client import LLMClient
       from shared.config import JUDGE_MODEL

       _PHRASING_PROMPT = (
           "Rephrase the user query in {n} different ways using different "
           "vocabulary while preserving the exact meaning. Cover synonyms "
           "and adjacent concepts (e.g. 'betrayal' -> 'duplicity', "
           "'self-interest', 'feigned loyalty'). Reply ONLY with a JSON "
           "object of the form: {{\"rephrasings\": [\"...\", \"...\", \"...\"]}}. "
           "Do not include the original. Do not add any commentary."
       )

       @dataclass(frozen=True)
       class MultiQueryResult:
           original_query: str
           phrasings: tuple[str, ...]            # excludes the original
           queries_used: tuple[str, ...]         # original + phrasings, in order
           merged_sections: tuple[tuple[int, WikiSection], ...]
           formatted: str
           rephrase_error: str | None            # populated if rephrase failed

       def _parse_phrasings(raw: str, n: int) -> list[str]:
           """Tolerant JSON parse. Strips code fences, accepts both
           {"rephrasings": [...]} and a bare list. Returns up to n
           non-empty stripped strings. Returns [] on any failure."""
           # implementation: try json.loads; if that fails, find first '[' and
           # last ']' and try again; coerce to list[str]; strip and filter
           # empty; cap to n.
           ...

       def _generate_phrasings(
           query: str, n: int, client: LLMClient
       ) -> tuple[list[str], str | None]:
           """Returns (phrasings, error_or_None). Single LLM call to
           JUDGE_MODEL. Caller does NOT retry on failure — the retrieve
           pipeline falls back to original-query-only with the error captured
           in MultiQueryResult.rephrase_error."""
           messages = [{
               "role": "user",
               "content": _PHRASING_PROMPT.format(n=n) + f"\n\nQuery: {query}",
           }]
           try:
               raw = client.generate(
                   messages,
                   model=JUDGE_MODEL,
                   temperature=0.3,
                   max_tokens=400,
                   purpose="multi_query_rephrase",
                   response_format={"type": "json_object"},
               )
           except Exception as exc:
               return [], f"rephrase_call_failed: {exc!r}"
           parsed = _parse_phrasings(raw, n)
           if not parsed:
               return [], "rephrase_parse_failed: empty list"
           return parsed, None

       def multi_query_retrieve(
           query: str,
           *,
           n: int = 3,
           client: LLMClient | None = None,
       ) -> MultiQueryResult:
           """Public entry point. Always returns a MultiQueryResult — never
           raises. On rephrase failure, falls back to original query only
           and records the error string in `rephrase_error`."""
           # 1. Generate phrasings (or fail-safe to original-only).
           # 2. queries_used = [query, *phrasings]
           # 3. For each q: retrieve_sections(q). Union by (page_rel, section_title).
           #    Keep MAX score per unique section.
           # 4. Sort merged sections by (-score, page_rel, section_title).
           # 5. formatted = format_sections(merged_sorted).
           # 6. Return MultiQueryResult.

   IMPORTANT: do NOT import from scripts/. Multi-query is library code; the
   harness imports it, not the other way around.
